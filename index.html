Mở trực tiếp trong Safari
const $ = id => document.getElementById(id); $('open-direct').href = EXEC_URL; function show(id){ ['loading','offline','slow'].forEach(k => $(k).hidden = k !== id); } function start(){ if (!navigator.onLine){ show('offline'); return; } show('loading'); const f = $('app'); let done = false; f.onload = () => { done = true; ['loading','offline','slow'].forEach(k => $(k).hidden = true); }; f.src = EXEC_URL; setTimeout(() => { if (!done) show('slow'); }, 20000); // quá 20 giây chưa mở được → gợi ý thử lại / mở trực tiếp } window.addEventListener('online', () => { if (!$('offline').hidden) start(); }); window.addEventListener('offline', () => show('offline')); start(); // Service worker: chỉ lưu vỏ ứng dụng (icon, trang chờ) để mở nhanh; KHÔNG lưu dữ liệu báo giá if ('serviceWorker' in navigator) navigator.serviceWorker.register('sw.js').catch(() => {});
